    def _parse_terraform_outputs(raw: dict) -> dict:
        """Parse terraform output -json into structured format for Firestore (D-04).

        Input format: {"output_name": {"sensitive": bool, "type": ..., "value": ...}, ...}
        Output format: {
            "outputs": {"output_name": value, ...},
            "by_node": {"node_name": {"attr": value, "node_type": "..."}, ...}
        }
        """
        flat = {}
        by_node = {}
        for key, entry in raw.items():
            value = entry.get("value") if isinstance(entry, dict) else entry
            flat[key] = value
            # Parse node_name from output key: "my-service_endpoint" -> node="my-service", attr="endpoint"
            # Split on last underscore that creates a valid attr name
            parts = key.rsplit("_", 1)
            if len(parts) == 2:
                node_name, attr = parts
                if node_name not in by_node:
                    by_node[node_name] = {}
                by_node[node_name][attr] = value
        return {"outputs": flat, "by_node": by_node}
    ```

        ref = _project_ref(tenant_id, project_ref_id).collection("env_outputs").document(env_name)
        doc = ref.get()
        if not doc.exists:

        data = doc.to_dict()

    @mcp.tool()
    def get_deploy_outputs(
        project_ref_id: str,
        env_name: str = "dev",
    ) -> str:
        """Get the latest Terraform outputs for a deployed environment.

        Returns structured output values (URLs, IPs, connection strings) from
        the most recent successful deploy. Use this to discover what resources
        are running and how to connect to them.

        Args:
            project_ref_id: The Firestore document ID of the project.
            env_name: The environment to query (default: "dev").
        """
        try:
            _ensure_deps()
            tenant_id = get_tenant_id()

            doc = deploy_store.get_env_outputs(tenant_id, project_ref_id, env_name)
            if not doc:
                return format_error(
                    "not_found",
                    f"No outputs found for environment '{env_name}'. "
                    "Deploy first, then outputs will be available.",
                )
            return format_json_response(doc)
        except Exception as e:
            log.error("get_deploy_outputs failed: %s", e)
            return format_error("get_deploy_outputs_error", str(e))
    ```

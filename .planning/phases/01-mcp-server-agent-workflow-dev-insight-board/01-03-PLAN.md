    2. Create `apps/web/src/stores/insight-store.ts`:
       Zustand store following existing pattern from auth-store.ts and blueprint-store.ts.

       ```typescript
       import { create } from 'zustand'
       import { doc, collection, onSnapshot, query, orderBy, limit, type Unsubscribe } from 'firebase/firestore'
       import { db } from '@/lib/firebase'

       interface InsightState {
         // State
         tenantId: string | null
         projectRefId: string | null
         env: string
         project: Record<string, any> | null
         nodes: Record<string, any>[]
         validation: Record<string, any> | null
         activity: Record<string, any>[]
         deployStatus: Record<string, any>[]
         loading: boolean
         connected: boolean
         error: string | null

         // Actions
         subscribe: (tenantId: string, projectRefId: string, env: string) => void
         setEnv: (env: string) => void
         cleanup: () => void
       }
       ```

       The `subscribe` action:
       - Stores tenantId, projectRefId, env in state
       - Creates 4 Firestore onSnapshot subscriptions:
         a. Project doc: `doc(db, 'tenants', tenantId, 'projects', projectRefId)` -> updates `project`
         b. Nodes collection: `collection(db, 'tenants', tenantId, 'projects', projectRefId, 'blueprints', env, 'nodes')` -> updates `nodes`
         c. Validation doc: `doc(db, 'tenants', tenantId, 'projects', projectRefId, 'blueprints', env, 'validation', 'current')` -> updates `validation`
         d. Activity collection: `query(collection(db, 'tenants', tenantId, 'projects', projectRefId, 'activity'), orderBy('timestamp', 'desc'), limit(50))` -> updates `activity`
       - Stores unsubscribe functions in a module-level array
       - Sets `connected: true` after first snapshot, `loading: false`
       - On snapshot error: sets `connected: false`, `error: "Connection lost..."`

       The `setEnv` action:
       - Calls cleanup() on existing subscriptions
       - Calls subscribe() with new env (re-subscribes nodes, validation for new env)

       The `cleanup` action:
       - Calls all stored unsubscribe functions
       - Resets state to initial values

       import { useEffect } from 'react'
       import { useParams } from 'react-router-dom'
       import { useInsightStore } from '@/stores/insight-store'
       import { useAuthStore } from '@/stores/auth-store'

       export default function InsightBoard() {
         const { projectRefId } = useParams<{ projectRefId: string }>()
         const user = useAuthStore(s => s.user)
         const { subscribe, cleanup, project, env, setEnv, loading, connected, error } = useInsightStore()

         useEffect(() => {
           if (user?.uid && projectRefId) {
             subscribe(user.uid, projectRefId, 'dev')
           }
           return cleanup
         }, [user?.uid, projectRefId])

         if (loading) {
           return <div className="min-h-screen bg-background flex items-center justify-center">
             <div className="text-muted-foreground">Loading insight board...</div>
           </div>
         }

         return (
           <div className="min-h-screen bg-background text-foreground">
             {/* Connection banner */}
             {!connected && error && (
               <div className="bg-amber-500/10 border-b border-amber-500/20 px-8 py-2 text-sm text-amber-600">
                 {error}
               </div>
             )}

             {/* TopBar */}
             <header className="border-b px-8 py-4 flex items-center justify-between">
               <h1 className="text-lg font-semibold">
                 {project?.display_name || 'Loading...'} — Insight Board
               </h1>
               {/* Environment tabs */}
               <div className="flex gap-2">
                 {(project?.environments || []).map((e: any) => (
                   <button
                     key={e.name}
                     onClick={() => setEnv(e.name)}
                     className={`px-3 py-1 rounded-md text-sm ${

                         : 'bg-muted text-muted-foreground hover:bg-muted/80'

                   >
                     {e.name}
                   </button>
                 ))}
               </div>
             </header>

               <div className="flex flex-col gap-6">
                 <div className="rounded-lg border bg-card p-4 text-card-foreground">
                   <p className="text-sm text-muted-foreground">Project State Panel — coming in Plan 06</p>
                 </div>
                 <div className="rounded-lg border bg-card p-4 text-card-foreground">
                   <p className="text-sm text-muted-foreground">Resource List Panel — coming in Plan 06</p>
                 </div>
                 <div className="rounded-lg border bg-card p-4 text-card-foreground">
                   <p className="text-sm text-muted-foreground">Validation Panel — coming in Plan 06</p>
